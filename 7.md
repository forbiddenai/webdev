```
<!DOCTYPE html>
<html>
<head>
    <title>Dynamic DOM Content</title>
    <style>
        #contentArea {
            margin-top: 20px;
            padding: 15px;
            border: 2px dashed #007bff;
            min-height: 100px;
            background-color: #f8f9fa;
        }
        .dynamic-item {
            background-color: #e9ecef;
            padding: 10px;
            margin-bottom: 10px;
            border-radius: 5px;
            border-left: 5px solid #007bff;
        }
    </style>
    <script>
        let counter = 1;

        function addContent() {
            // Get the container element
            let container = document.getElementById('contentArea');
            
            // Create a new div element
            let newDiv = document.createElement('div');
            newDiv.className = 'dynamic-item';
            
            // Create text node for content
            let contentText = document.createTextNode('Dynamic Element #' + counter + ' created at ' + new Date().toLocaleTimeString());
            
            // Append text to div
            newDiv.appendChild(contentText);
            
            // Append div to container
            container.appendChild(newDiv);
            
            counter++;
        }

        function clearContent() {
            document.getElementById('contentArea').innerHTML = '';
            counter = 1;
        }
    </script>
</head>
<body>
    <h2>Dynamic Content using DOM</h2>
    <p>Click the buttons below to dynamically add or remove elements using JavaScript DOM manipulation.</p>
    
    <button onclick="addContent()" style="padding: 8px 15px; cursor: pointer;">Add Element</button>
    <button onclick="clearContent()" style="padding: 8px 15px; cursor: pointer; color: red;">Clear All</button>
    
    <div id="contentArea"></div>
</body>
</html>
```
